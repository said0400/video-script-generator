# ══════════════════════════════════════════════════════════════════════════════
# 📝 Migration Notes (WhisperX → stable-ts)
# ══════════════════════════════════════════════════════════════════════════════
#
# Old (WhisperX):
#   import whisperx
#   model        = whisperx.load_model("medium", device="cpu")
#   result       = model.transcribe(audio)
#   align_model  = whisperx.load_align_model(...)
#   aligned      = whisperx.align(result, align_model, ...)
#
# New (stable-ts):
#   import stable_whisper
#   model  = stable_whisper.load_faster_whisper("medium", device="cpu")
#   result = model.transcribe_stable(audio, language="ar", word_timestamps=True)
#   # alignment built-in, no separate align model needed!
#
# Benefits:
#   ✅ One library instead of two
#   ✅ No pyannote.audio dependency
#   ✅ No HF_TOKEN needed
#   ✅ Active maintenance
#   ✅ Better Arabic support
#   ✅ Faster setup
